  <script>
    // TEMPEL URL WEB APP GOOGLE APPS SCRIPT ANDA DI SINI
    const API_URL = "https://script.google.com/macros/s/AKfycbwWO7YWx99HKZ5Mo9sWqnnyBNBITCeru-8XdZQL-pSK.../exec";

    let selectedBusSeat = '';
    let selectedJeepSeat = '';
    let studentName = '';
    let typingTimer = null;

    function switchTab(tab) {
      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
      if (tab === 'siswa') {
        document.querySelectorAll('.tab-btn')[0].classList.add('active');
        document.getElementById('siswaView').style.display = 'block';
        document.getElementById('adminView').style.display = 'none';
      } else {
        document.querySelectorAll('.tab-btn')[1].classList.add('active');
        document.getElementById('siswaView').style.display = 'none';
        document.getElementById('adminView').style.display = 'block';
        loadManifest();
      }
    }

    function debounceFetchStudent() {
      clearTimeout(typingTimer);
      typingTimer = setTimeout(fetchStudent, 400);
    }

    function fetchStudent() {
      let tkt = document.getElementById('noTiket').value.trim();
      let msg = document.getElementById('ticketMsg');
      if (tkt.length < 3) { resetForm(); return; }
      msg.style.color = '#3b82f6';
      msg.innerText = '🔍 Mencari data tiket...';
      
      fetch(`${API_URL}?action=getStudent&ticket=${encodeURIComponent(tkt)}`)
        .then(res => res.json())
        .then(res => {
          if (res.success) {
            studentName = res.nama;
            document.getElementById('namaSiswa').value = res.nama;
            document.getElementById('kelasSiswa').value = res.kelas;
            document.getElementById('busSelect').disabled = false;
            document.getElementById('jeepSelect').disabled = false;
            msg.style.color = '#10b981';
            msg.innerText = '✓ Tiket Terverifikasi';

            if (res.bus) {
              document.getElementById('busSelect').value = res.bus;
              selectedBusSeat = res.seatBus;
              loadBusSeats();
            }
            if (res.jeep) {
              document.getElementById('jeepSelect').value = res.jeep;
              selectedJeepSeat = res.seatJeep;
              loadJeepSeats();
            }
          } else {
            resetForm();
            msg.style.color = '#ef4444';
            msg.innerText = res.message;
          }
        });
    }

    function resetForm() {
      studentName = '';
      document.getElementById('namaSiswa').value = '';
      document.getElementById('kelasSiswa').value = '';
      document.getElementById('busSelect').disabled = true;
      document.getElementById('jeepSelect').disabled = true;
      document.getElementById('busSelect').value = '';
      document.getElementById('jeepSelect').value = '';
      document.getElementById('busSeatContainer').innerHTML = '';
      document.getElementById('jeepSeatContainer').innerHTML = '';
      selectedBusSeat = '';
      selectedJeepSeat = '';
      updateSummary();
    }

    function loadBusSeats() {
      let bus = document.getElementById('busSelect').value;
      let container = document.getElementById('busSeatContainer');
      if (!bus) { container.innerHTML = ''; return; }

      container.innerHTML = '<div style="padding:10px; font-size:12px;">Memuat denah kursi...</div>';
      fetch(`${API_URL}?action=getOccupied&vehicle=${encodeURIComponent(bus)}`)
        .then(res => res.json())
        .then(occ => {
          container.innerHTML = '';
          container.className = 'seat-grid-bus';
          for (let i = 1; i <= 52; i++) {
            let sName = 'Seat ' + i;
            let el = document.createElement('div');
            el.className = 'seat-item';
            
            if (occ[sName] && occ[sName] !== studentName) {
              el.classList.add('occupied');
              el.innerHTML = `${sName}<br><small style="font-size:9px;">(${occ[sName]})</small>`;
            } else {
              el.innerText = sName;
              if (sName === selectedBusSeat) el.classList.add('selected');
              
              el.onclick = () => {
                if (!studentName) {
                  alert('⚠️ Silakan masukkan Nomor Tiket yang valid terlebih dahulu!');
                  return;
                }
                container.querySelectorAll('.seat-item').forEach(b => b.classList.remove('selected'));
                el.classList.add('selected');
                selectedBusSeat = sName;
                updateSummary();
              };
            }
            container.appendChild(el);
          }
        });
    }

    function loadJeepSeats() {
      let jeep = document.getElementById('jeepSelect').value;
      let container = document.getElementById('jeepSeatContainer');
      if (!jeep) { container.innerHTML = ''; return; }

      container.innerHTML = '<div style="padding:10px; font-size:12px;">Memuat denah kursi...</div>';
      fetch(`${API_URL}?action=getOccupied&vehicle=${encodeURIComponent(jeep)}`)
        .then(res => res.json())
        .then(occ => {
          container.innerHTML = '';
          container.className = 'seat-grid-jeep';
          for (let i = 1; i <= 5; i++) {
            let sName = 'Kursi ' + i;
            let el = document.createElement('div');
            el.className = 'seat-item';

            if (occ[sName] && occ[sName] !== studentName) {
              el.classList.add('occupied');
              el.innerHTML = `${sName}<br><small style="font-size:9px;">(${occ[sName]})</small>`;
            } else {
              el.innerText = sName;
              if (sName === selectedJeepSeat) el.classList.add('selected');

              el.onclick = () => {
                if (!studentName) {
                  alert('⚠️ Silakan masukkan Nomor Tiket yang valid terlebih dahulu!');
                  return;
                }
                container.querySelectorAll('.seat-item').forEach(b => b.classList.remove('selected'));
                el.classList.add('selected');
                selectedJeepSeat = sName;
                updateSummary();
              };
            }
            container.appendChild(el);
          }
        });
    }

    function updateSummary() {
      let busVal = document.getElementById('busSelect').value;
      let jeepVal = document.getElementById('jeepSelect').value;

      document.getElementById('sumBus').innerText = busVal && selectedBusSeat ? `${busVal} - ${selectedBusSeat}` : 'Belum Dipilih';
      document.getElementById('sumJeep').innerText = jeepVal && selectedJeepSeat ? `${jeepVal} - ${selectedJeepSeat}` : 'Belum Dipilih';

      let btn = document.getElementById('submitBtn');
      if (studentName && ((busVal && selectedBusSeat) || (jeepVal && selectedJeepSeat))) {
        btn.disabled = false;
        btn.classList.add('ready');
      } else {
        btn.disabled = true;
        btn.classList.remove('ready');
      }
    }

    function saveAll() {
      let tkt = document.getElementById('noTiket').value.trim();
      let nama = document.getElementById('namaSiswa').value.trim();
      let kelas = document.getElementById('kelasSiswa').value.trim();
      let bus = document.getElementById('busSelect').value;
      let jeep = document.getElementById('jeepSelect').value;

      let btn = document.getElementById('submitBtn');
      btn.innerText = 'Menyimpan...';
      btn.disabled = true;

      fetch(API_URL, {
        method: 'POST',
        body: JSON.stringify({
          ticketNo: tkt,
          nama: nama,
          kelas: kelas,
          busName: bus,
          seatBus: selectedBusSeat,
          jeepName: jeep,
          seatJeep: selectedJeepSeat
        })
      })
      .then(res => res.json())
      .then(res => {
        if (res.success) {
          alert(`BERHASIL DISIMPAN!\n\nNama: ${res.nama}\nBus: ${res.bus} (${res.seatBus})\nJeep: ${res.jeep} (${res.seatJeep})`);
          document.getElementById('noTiket').value = '';
          document.getElementById('ticketMsg').innerText = '';
          btn.innerText = 'Simpan Pilihan Bus & Jeep';
          resetForm();
        } else {
          alert(res.message);
          btn.innerText = 'Simpan Pilihan Bus & Jeep';
          btn.disabled = false;
        }
      });
    }

    function loadManifest() {
      fetch(`${API_URL}?action=getManifest`)
        .then(res => res.json())
        .then(data => {
          let tbody = document.getElementById('manifestTable');
          if (data.length === 0) {
            tbody.innerHTML = '<tr><td colspan="5" style="text-align:center;">Belum ada data.</td></tr>';
            return;
          }
          tbody.innerHTML = data.map(d => `
            <tr>
              <td><b>${d.noTiket}</b></td>
              <td>${d.nama}</td>
              <td>${d.kelas}</td>
              <td><span style="color:#2563eb; font-weight:bold;">${d.bus}</span></td>
              <td><span style="color:#059669; font-weight:bold;">${d.jeep}</span></td>
            </tr>
          `).join('');
        });
    }
  </script>
